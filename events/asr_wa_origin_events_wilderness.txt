country_event = {
  id = asr_wa_wilderness.10
  hide_window = yes
  is_triggered_only = yes
  trigger = { is_wilderness_empire = yes }

  # 将开局的生物质库存设置为正确的值
  # 4.2某个版本之后, 荒野开局的生物质库存始终为0, 这导致即使通过贸易市场购买到足够的资源, 也无法在在第一个月就开始建设

    set_resource = {
      resource = biomass
      value = trigger:asr_wa_wilderness_pop_amount
    }
  }
}

country_event = {
  id = asr_wa_wilderness.100
  hide_window = yes
  is_triggered_only = yes
  trigger = { is_wilderness_empire = yes }

  immediate = {
    # 系统会给高级AI添加额外的开局区划, 然而添加给荒野的并不是共栖群落而是蜂巢的蜂巢集群
    # 月底检测的时候会把共栖群落和蜂巢集群合并, 然而并不是合并到共栖群落而是合并到蜂巢集群
    # 这会导致共栖群落建筑槽中的起始建筑全部被清空, 蜂巢集群建筑槽中的建筑才能留保留下来
    # 当然蜂巢集群里面本身不会有任何建筑, 并且蜂巢集群最后会被转换成共栖群落
    # 所以约等于过月直接清空荒野的所有建筑
    # 4.4版本已修复
    every_owned_colony = {
      # 这里通过手动移除蜂巢集群来避免触发区划合并导致建筑被清空
      # 然而殖民地添加的是城市区划和蜂巢集群, 甚至没有共栖群落
      # 这会导致最后的一个城市区划无法被移除, 无法避免触发区划合并
      # 只能在移除后手动添加被移除的建筑
      asr_wa_change_to_wilderness_districts = yes
      # 立即触发一次检测, 如果触发区划合并的话还是会丢建筑
      validate_planet_buildings_and_districts = yes
      # 手动添加被移除的建筑
      asr_wa_restore_wilderness_buildings = yes
    }
  }
}

country_event = {
  id = asr_wa_wilderness.200
  hide_window = yes
  is_triggered_only = yes
  trigger = {
    is_wilderness_empire = yes
    has_tradition = tr_purity_finish
    any_owned_species = {
      OR = {
        has_trait = trait_wilderness
        # 更新非荒野人口 (如果有其他Mod可以让荒野利用非荒野人口)
        is_archetype = PRESAPIENT
        is_archetype = BIOLOGICAL
        is_archetype = LITHOID
      }
    }
  }

  immediate = {
    # 更新所有物种, 这样就可以在事件给物种添加上特质时用这种方式刷新变量
    # from = { set_species_traits_purity_count_variable = yes }
    every_owned_species = {
      limit = {
        OR = {
          has_trait = trait_wilderness
          is_archetype = PRESAPIENT
          is_archetype = BIOLOGICAL
          is_archetype = LITHOID
        }
      }
      set_species_traits_purity_count_variable = yes
    }
  }
}

# See: plant.115
colony_event = {
  id = asr_wa_wilderness.2010
  hide_window = yes
  is_triggered_only = yes

  trigger = {
    owner? = {
      is_wilderness_empire = yes
      is_idyllic_bloom_empire = yes
      has_policy_flag = asr_wa_p_idyllic_bloom_valid
    }
    asr_wa_is_gaia_planet = yes
  }

  immediate = {
    owner = { set_country_flag = has_bloomed_pops }
    every_owned_pop_group = {
      limit = {
        species = { has_trait = trait_wilderness }
        OR = {
          species = { is_flora_species = yes }
          AND = {
            species = { is_robotic_species = no }
            root.owner = { has_tradition = tr_mutation_unnatural_selection }
          }
        }
        is_being_assimilated = no
        is_being_purged = no
        NOR = {
          has_trait = trait_plantoid_bloomed
          has_trait = trait_malleable_genes
        }
      }
      save_event_target_as = asr_wa_pre_bloom_pop_group
      species = { save_event_target_as = asr_wa_pre_bloom_species }

      if = {
        limit = {
          any_galaxy_species = {
            has_species_flag = bloomed_species_of_@event_target:asr_wa_pre_bloom_species
          }
        }
        random_galaxy_species = {
          limit = {
            has_species_flag = bloomed_species_of_@event_target:asr_wa_pre_bloom_species
          }
          save_event_target_as = asr_wa_post_bloom_species
        }
      } else = {
        species = {
          create_species = {
            is_mod = yes  # 创建的物种是否视为当前作用域物种的子物种
            # mod_name_affix = trait_plantoid_bloomed  # 新物种的修饰词 (自己填的好像默认为前缀), 不填则随机 (例如: <name>变体, <name>异体, 高<name>)
            name = this
            plural = this
            class = this
            portrait = this
            traits = this
            can_be_modified = this
            homeworld = this
            namelist = this
            gender = this
            traits = { trait = trait_plantoid_bloomed }
            effect = {
              if = {
                limit = {
                  NOR = {
                    has_trait = trait_auto_hab_preference
                    has_trait = trait_pc_gaia_preference
                    has_trait = trait_pc_gaia_preference_terraforming
                  }
                }
                set_habitability_trait = { trait = trait_pc_gaia_preference }
              }
              save_event_target_as = asr_wa_post_bloom_species
              set_species_flag = bloomed_species_of_@event_target:asr_wa_pre_bloom_species
            }
          }
        }
      }

      colony = {
        create_pop_group = {
          pop_group = event_target:asr_wa_pre_bloom_pop_group
          species = event_target:asr_wa_post_bloom_species
          size = 0
          effect = {
            transfer_pop_amount = {
              source = event_target:asr_wa_pre_bloom_pop_group
              target = this
              percentage = 1
            }
          }
        }
      }

    }
  }
}
