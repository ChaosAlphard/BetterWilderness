# 护卫哨所
building_asr_wa_sentinel_posts = {
  icon = "building_sentinel_posts"
  category = government
  base_buildtime = @b1_time
  can_build = yes
  can_demolish = yes
  can_be_ruined = yes

  building_sets = {
    government
    urban
  }

  inline_script = {
    script = buildings/wilderness_buildings_potential_destroy
  }
  allow = {
    hidden_trigger = {
      OR = {
        owner = { is_ai = no }
        AND = {
          NOT = { has_forbidden_jobs = patrol_drone }
          NOT = { has_available_jobs = patrol_drone }
        }
      }
    }
  }

  resources = {
    category = planet_buildings
    inline_script = {
      script = asr_wa/building/shared_cost
      BIOMASS = @b2_biomass
      FOOD_AND_MINERAL_COST = @wa_b1_mixed
      FOOD_COST = @wa_b1_foods
      MINERAL_COST = @wa_b1_minerals
    }
    upkeep = {
      food = @b1_upkeep
    }
  }

  planet_modifier = {
    job_patrol_drone_per_pop = 0.02
  }
  # AI无法识别按人口数提供的岗位, 所以需要有固定提供的岗位让AI识别
  triggered_planet_modifier = {
    potential = {
      owner? = { is_ai = yes }
    }
    job_patrol_drone_add = 200
  }

  prerequisites = {
    "tech_planetary_government"
  }

  show_tech_unlock_if = {
    always = no
  }

  inline_script = {
    script = buildings/on_all_wilderness_buildings_districts
  }

  ai_estimate_without_unemployment = yes
}

building_asr_wa_archaeo_refinery = {
  icon = "building_archaeo_refinery"
  ruined_icon = "building_ruined"
  base_buildtime = @buildtime_4
  category = manufacturing
  can_build = yes
  can_demolish = yes
  planet_limit = 1

  building_sets = {
    government
    industrial
    foundry
    factory
  }

  potential = {
    owner? = {
      is_wilderness_empire = yes
      is_ai = yes
      has_technology = tech_archaeostudies
    }
  }
  allow = {
    owner? = {
      is_wilderness_empire = yes
      is_ai = yes
      has_technology = tech_archaeostudies
    }
  }
  destroy_trigger = {
    owner? = {
      is_ai = no
    }
  }

  resources = {
    category = planet_buildings
    inline_script = {
      script = asr_wa/building/shared_cost
      BIOMASS = @b2_biomass
      FOOD_AND_MINERAL_COST = @wa_b2_mixed
      FOOD_COST = @wa_b2_foods
      MINERAL_COST = @wa_b2_minerals
    }
    cost = {
      minor_artifacts = @b4_rare_cost
    }
    upkeep = {
      energy = @b1_upkeep
    }
  }

  triggered_planet_modifier = {
    potential = {
      owner? = { is_ai = yes }
    }
    planet_metallurgists_exotic_gases_produces_add = 0.1
    planet_metallurgists_rare_crystals_produces_add = 0.1
    planet_metallurgists_volatile_motes_produces_add = 0.1
  }
  triggered_planet_modifier = {
    potential = {
      owner? = {
        is_ai = yes
        has_ascension_perk = ap_archaeoengineers
      }
    }
    planet_metallurgists_exotic_gases_produces_add = 0.05
    planet_metallurgists_rare_crystals_produces_add = 0.05
    planet_metallurgists_volatile_motes_produces_add = 0.05
  }
  triggered_planet_modifier = {
    potential = {
      owner? = {
        is_ai = yes
        is_catalytic_empire = no
      }
    }
    planet_metallurgists_minerals_upkeep_add = 1
  }
  triggered_planet_modifier = {
    potential = {
      owner? = {
        is_ai = yes
        is_catalytic_empire = yes
      }
    }
    planet_metallurgists_food_upkeep_add = 1.5
  }
}
